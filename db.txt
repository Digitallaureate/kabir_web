import Layout from '../components/layout/Layout.astro';

import Button from '../components/ui/Button.astro';
import { getFirestore, collection, getDocs, query, where, orderBy } from "firebase/firestore";
import { getAuth, onAuthStateChanged } from "firebase/auth";
import { app } from "../utils/firebase";

export const prerender = false;

// Initialize Firestore
const db = getFirestore(app);

// Check if user is logged in (server-side check)
const auth = getAuth(app);
let isLoggedIn = false;

// For server-side rendering, we'll fetch all data and filter on the client side
// Fetch cities from Firestore
const citiesSnapshot = await getDocs(query(collection(db, "cities"), where("is_active", "==", true), orderBy("name")));
const cities = [];
citiesSnapshot.forEach((doc) => {
  cities.push({
    id: doc.id,
    name: doc.data().name || 'Unknown City',
    description: doc.data().description || '',
    location: doc.data().location || '',
    latitude: parseFloat(doc.data().latitude) || 0,
    longitude: parseFloat(doc.data().longitude) || 0,
    zoomLevel: parseInt(doc.data().zoom_level) || 10,
    period: doc.data().period || ''
  });
});

// Fetch monuments from Firestore
const monumentsSnapshot = await getDocs(query(collection(db, "monuments"), orderBy("name")));
const monuments = [];
monumentsSnapshot.forEach((doc) => {
  monuments.push({
    id: doc.id,
    name: doc.data().name || 'Unknown Monument',
    description: doc.data().description || '',
    category: doc.data().category || 'Unknown',
    period: doc.data().period || '',
    yearBuilt: doc.data().year_built || '',
    architect: doc.data().architect || '',
    latitude: parseFloat(doc.data().latitude) || 0,
    longitude: parseFloat(doc.data().longitude) || 0,
    cityId: doc.data().city_id || '',
    address: doc.data().address || '',
    isActive: doc.data().is_active || false,
    imageUrl: doc.data().image_url || '/assets/images/monument-placeholder.jpg'
  });
});

// Fetch sub-monuments from Firestore
const subMonumentsSnapshot = await getDocs(query(collection(db, "sub_monuments"), orderBy("name")));
const subMonuments = [];
subMonumentsSnapshot.forEach((doc) => {
  subMonuments.push({
    id: doc.id,
    parentId: doc.data().parent_id || '',
    name: doc.data().name || 'Unknown Point of Interest',
    title: doc.data().title || '',
    description: doc.data().description || '',
    content: doc.data().content || '',
    category: doc.data().category || 'Unknown',
    latitude: parseFloat(doc.data().latitude) || 0,
    longitude: parseFloat(doc.data().longitude) || 0,
    isActive: doc.data().is_active || false,
    imageUrl: doc.data().image_url || '/assets/images/submonument-placeholder.jpg',
    tags: doc.data().tags || []
  });
});

// Fetch routes from Firestore
const routesSnapshot = await getDocs(query(collection(db, "routes"), orderBy("name")));
const routes = [];
routesSnapshot.forEach((doc) => {
  routes.push({
    id: doc.id,
    name: doc.data().name || 'Unnamed Route',
    description: doc.data().description || '',
    monumentIds: doc.data().monument_ids || [],
    order: doc.data().order || [],
    type: doc.data().type || 'custom',
    estimatedTime: doc.data().estimated_time || '',
    distance: doc.data().distance || '',
    isActive: doc.data().is_active || false,
    color: doc.data().color || '#c2634b'
  });
});

// Default to Agra center
const defaultCenter = { lat: 27.1767, lng: 78.0081 };

<script>
  // Helper functions
  function getCityById(id) {
    return allCities.find(city => city.id === id);
  }
  
  function getMonumentById(id) {
    return allMonuments.find(monument => monument.id === id);
  }
  
  function getRouteById(id) {
    return allRoutes.find(route => route.id === id);
  }
  
  function getSubMonumentsByParentId(parentId) {
    return allSubMonuments.filter(subMonument => subMonument.parentId === parentId);
  }
  
  // This script will handle the map functionality using Google Maps API
  document.addEventListener('DOMContentLoaded', function() {
    // Check authentication state
    const authCheck = document.getElementById('auth-check');
    let isLoggedIn = authCheck ? authCheck.dataset.isLoggedIn === 'true' : false;
    
    // Perform client-side auth check (for SPA navigation)
    const auth = firebase.auth();
    
    auth.onAuthStateChanged((user) => {
      if (user) {
        isLoggedIn = true;
        initializeMap();
      } else {
        isLoggedIn = false;
        showLoginModal();
      }
    });
    
    // Parse data from server
    const allCities = JSON.parse(`${JSON.stringify(cities)}`);
    const allMonuments = JSON.parse(`${JSON.stringify(monuments)}`);
    const allSubMonuments = JSON.parse(`${JSON.stringify(subMonuments)}`);
    const allRoutes = JSON.parse(`${JSON.stringify(routes)}`);
    const defaultMapCenter = JSON.parse(`${JSON.stringify(defaultCenter)}`);
    
    // Initialize UI elements
    initUI();
    
    // Login modal
    const loginModal = document.getElementById('login-required-modal');
    const modalClose = document.querySelector('.modal-close');
    
    function showLoginModal() {
      if (loginModal) {
        loginModal.classList.add('active');
      }
    }
    
    if (modalClose) {
      modalClose.addEventListener('click', () => {
        loginModal.classList.remove('active');
      });
    }
    
    let map;
    let markers = {};
    let polylines = [];
    let activeInfoWindow = null;
    let currentZoomLevel = 'city'; // 'city', 'monument', or 'detail'
    
    function initializeMap() {
      const mapOptions = {
        center: defaultMapCenter,
        zoom: 10,
        mapTypeId: google.maps.MapTypeId.ROADMAP,
        mapTypeControl: false,
        streetViewControl: false,
        fullscreenControl: true,
        zoomControl: true,
        styles: [
          {
            "elementType": "geometry",
            "stylers": [
              {
                "color": "#ebe3cd"
              }
            ]
          },
          {
            "elementType": "labels.text.fill",
            "stylers": [
              {
                "color": "#523735"
              }
            ]
          },
          {
            "elementType": "labels.text.stroke",
            "stylers": [
              {
                "color": "#f5f1e6"
              }
            ]
          },
          {
            "featureType": "administrative",
            "elementType": "geometry.stroke",
            "stylers": [
              {
                "color": "#c9b2a6"
              }
            ]
          },
          {
            "featureType": "administrative
